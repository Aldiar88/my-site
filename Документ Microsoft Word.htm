    <script>

        // Tab Navigation Logic

        function switchTab(tabId) {

            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));

            document.querySelectorAll('.tab-btn').forEach(btn => {

                btn.classList.remove('text-emerald-400', 'bg-slate-800');

                btn.classList.add('text-slate-300');

                if(btn.dataset.target === tabId) {

                    btn.classList.add('text-emerald-400', 'bg-slate-800');

                    btn.classList.remove('text-slate-300');

                }

            });

            document.getElementById(tabId).classList.add('active');

            if(tabId === 'tab-d') map.invalidateSize();

        }

 

        // --- MATH ENGINE & CONSTANTS ---

        const T_surface_avg = 30.0;

        const A_0 = 15.0; // Amplitude

        const alpha = 0.043; // m2/day

        const tau = 365;

 

        // SECTION A: Logic

        const tonnageIn = document.getElementById('tonnage');

        const depthIn = document.getElementById('depth');

        const cropIn = document.getElementById('crop-select');

       

        function updateSectionA() {

            document.getElementById('tonnage-val').innerText = tonnageIn.value;

            document.getElementById('depth-val').innerText = parseFloat(depthIn.value).toFixed(1);

           

            const z = parseFloat(depthIn.value);

            const tons = parseFloat(tonnageIn.value);

           

            // Fourier calculation for depth temp attenuation

            const attenuation = Math.exp(-z * Math.sqrt(Math.PI / (tau * alpha)));

            const currentT = 20.0 + (A_0 * attenuation); // Base ground temp offset + attenuated wave

           

            document.getElementById('out-temp').innerText = currentT.toFixed(1);

           

            // Panels proportional to volume/tonnage

            const panels = Math.ceil(tons * 0.3);

            document.getElementById('out-panels').innerText = panels;

           

            // Spoilage reduction and savings

            document.getElementById('out-spoilage').innerText = currentT < 18 ? '92' : '78';

           

            const savings = tons * 15000 + (z * 50000);

            document.getElementById('out-savings').innerText = savings.toLocaleString('ru-RU') + ' ₸';

        }

       

        tonnageIn.addEventListener('input', updateSectionA);

        depthIn.addEventListener('input', updateSectionA);

        cropIn.addEventListener('change', updateSectionA);

 

        // SECTION B: Logic

        const areaIn = document.getElementById('area');

        const wallIn = document.getElementById('wall');

        const domeIn = document.getElementById('dome');

 

        function updateSectionB() {

            document.getElementById('area-val').innerText = areaIn.value;

            document.getElementById('wall-val').innerText = wallIn.value;

            document.getElementById('dome-val').innerText = parseFloat(domeIn.value).toFixed(1);

           

            const wallT = parseFloat(wallIn.value);

            const height = parseFloat(domeIn.value);

           

            // Cellar depth based on wall inertia (simplified)

            const cellarZ = 2.0 / Math.max(0.2, wallT);

            document.getElementById('cellar-depth').innerText = cellarZ.toFixed(1) + ' м';

 

            // Bernoulli Draft (v = sqrt(2gh * dt/T))

            const v = Math.sqrt(2 * 9.81 * height * (5 / 300));

            document.getElementById('draft-speed').innerText = v.toFixed(2) + ' м/с';

           

            document.getElementById('temp-drop').innerText = (v * 2.8).toFixed(1) + ' °C';

        }

 

        areaIn.addEventListener('input', updateSectionB);

        wallIn.addEventListener('input', updateSectionB);

        domeIn.addEventListener('input', updateSectionB);

 

        // SECTION C: Logic

        const itLoadIn = document.getElementById('it-load');

 

        function updateSectionC() {

            const loadKw = parseFloat(itLoadIn.value);

            document.getElementById('it-load-val').innerText = loadKw;

 

            const pueTrad = 1.70;

            const pueGeo = 1.03;

           

            const hours = 8760;

            const energyTrad = loadKw * pueTrad * hours;

            const energyGeo = loadKw * pueGeo * hours;

            const savedKwh = energyTrad - energyGeo;

           

            document.getElementById('eco-kwh').innerText = savedKwh.toLocaleString('ru-RU') + ' кВт·ч';

            document.getElementById('eco-kzt').innerText = (savedKwh * 25).toLocaleString('ru-RU') + ' ₸';

            document.getElementById('eco-co2').innerText = (savedKwh * 0.7 / 1000).toLocaleString('ru-RU') + ' тонн';

        }

       

        itLoadIn.addEventListener('input', updateSectionC);

 

        // INITIALIZE ALL CALCULATORS

        updateSectionA();

        updateSectionB();

        updateSectionC();

 

        // CHART.JS Initialization (Section C)

        const ctx = document.getElementById('fourierChart').getContext('2d');

        const timeLabels = Array.from({length: 24}, (_, i) => i + ':00');

        const outWave = timeLabels.map((_, i) => 35 + 10 * Math.sin(i * Math.PI / 12 - 2));

        const inWave = timeLabels.map((_, i) => 24 + 1.5 * Math.sin(i * Math.PI / 12 - 6));

 

        new Chart(ctx, {

            type: 'line',

            data: {

                labels: timeLabels,

                datasets: [

                    { label: 'Т улицы (°C)', data: outWave, borderColor: '#f87171', tension: 0.4 },

                    { label: 'Т за стеной 0.5м (°C)', data: inWave, borderColor: '#34d399', tension: 0.4 }

                ]

            },

            options: {

                responsive: true,

                scales: {

                    y: { grid: { color: '#334155' }, ticks: { color: '#94a3b8' } },

                    x: { grid: { color: '#334155' }, ticks: { color: '#94a3b8' } }

                },

                plugins: { legend: { labels: { color: '#cbd5e1' } } }

            }

        });

 

        // LEAFLET.JS MAP Initialization (Section D)

        const map = L.map('map').setView([42.7, 69.5], 8);

        L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {

            attribution: '&copy; <a href="https://carto.com/">CARTO</a>'

        }).addTo(map);

 

        const geoData = [

            {

                coords: [42.9238, 69.7027], title: 'Пещера Ақмешіт',

                desc: 'Гравитационная стратификация холода (Cold Air Pool). Улица: 30.0°C. Дно: 15.0°C. Δρ = 0.059 кг/м³.'

            },

            {

                coords: [42.6025, 69.1719], title: 'Плато Ордабасы',

                desc: 'Затухание температурной волны в суглинках. Поверхность: 48.0°C, 15см: 22.0°C (ΔT=26.0°C).'

            },

            {

                coords: [43.2975, 68.2707], title: 'Түркістан (Мавзолей и Баня)',

                desc: 'Адиабатическое охлаждение (T_dry=23.9°C, T_wet=22.5°C). Вытяжка Бернулли над куполом.'

            },

            {

                coords: [42.4411, 70.4851], title: 'Каньон Кіші Қайыңды (Ақсу)',

                desc: 'Эффект Вентури (v=2.8 м/с). Градиент Г=9.76°C/км. Горный ручей снижает t°.'

            },

            {

                coords: [42.3417, 69.5901], title: 'Этномузей «Қылует» (Шымкент)',

                desc: 'Абсолютный подземный термостат (глубина 7м). Улица: 41.5°C, Внутри: 20.2°C (ΔT=21.3°C).'

            }

        ];

 

        geoData.forEach(site => {

            const popupContent = `

                <div style="color: #1e293b; font-family: sans-serif;">

                    <h3 style="font-weight: bold; margin-bottom: 5px; color: #059669;">${site.title}</h3>

                    <p style="font-size: 13px; line-height: 1.4; margin: 0;">${site.desc}</p>

                </div>

            `;

            L.circleMarker(site.coords, {

                radius: 8, fillColor: "#10b981", color: "#000", weight: 1, opacity: 1, fillOpacity: 0.8

            }).addTo(map).bindPopup(popupContent);

        });
